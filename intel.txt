            <label class="method"><input type="radio" name="method" value="alternative"><span><strong>Accessible alternative</strong><small>Continue with a non-biometric path in this example</small></span></label>
          </div>
          <label class="consent"><input id="demoConsent" type="checkbox"><span>I understand this is a fictional simulation and no real identity verification or personal-data processing will occur.</span></label>
          <div class="modal-actions"><button class="button primary" id="continueDemo">Continue with sample <span>→</span></button></div>
          <div class="disclaimer">Never enter or upload an ID, face image, fingerprint, password, or other sensitive information on this concept website.</div>
        `;
        document.getElementById('continueDemo').addEventListener('click', () => {
          if (!document.getElementById('demoConsent').checked) {
            showToast('Please confirm the demo notice to continue.');
            return;
          }
          const choice = document.querySelector('input[name="method"]:checked').value;
          renderReview(choice);
        });
      }

      function renderReview(choice) {
        const names = {
          credential: 'Trusted digital credential',
          passkey: 'Passkey / device authentication',
          alternative: 'Accessible non-biometric alternative'
        };
        body.innerHTML = `
          <div class="progress"><span class="active"></span><span class="active"></span><span></span></div>
          <div class="modal-kicker">Step 2 · Review the request</div>
          <h2 id="modalTitle">A sample service asks for a simple proof.</h2>
          <p class="modal-copy">Fictional purchase protection check · Requested proof: transaction eligibility · Selected path: <strong>${names[choice]}</strong></p>
          <div class="result-card"><strong>What this concept would share</strong>Only a result such as “eligible” — not a full identity profile or raw biometric data.</div>
          <div class="disclaimer">This is a user-interface demonstration. No service is contacted and no verification takes place.</div>
          <div class="modal-actions">
            <button class="button outline" id="backDemo">Back</button>
            <button class="button primary" id="approveDemo">Show sample result <span>→</span></button>
          </div>
        `;
        document.getElementById('backDemo').addEventListener('click', renderIntro);
        document.getElementById('approveDemo').addEventListener('click', () => renderResult(names[choice]));
      }

      function renderResult(method) {
        body.innerHTML = `
          <div class="progress"><span class="active"></span><span class="active"></span><span class="active"></span></div>
          <div class="modal-kicker">Step 3 · Sample outcome</div>
          <h2 id="modalTitle">Walkthrough complete.</h2>
          <p class="modal-copy">You explored the <strong>${method}</strong> path.</p>
          <div class="result-card"><strong>Sample status: eligible ✓</strong>This is invented demo content, not a real identity result. No personal data was submitted, stored, or checked.</div>
          <p class="modal-copy">A production service would require a secure backend, a vetted verification provider, threat modeling, legal and privacy review, accessibility testing, and ongoing operational monitoring.</p>
          <div class="modal-actions"><button class="button primary" id="doneDemo">Done</button></div>
        `;
        document.getElementById('doneDemo').addEventListener('click', closeDemo);
      }

      function openDemo(event) {
        if (event) event.preventDefault();
        lastFocus = document.activeElement;
        modal.classList.add('open');
        document.body.style.overflow = 'hidden';
        renderIntro();
        document.getElementById('closeModal').focus();
      }

      function closeDemo() {
        modal.classList.remove('open');
        document.body.style.overflow = '';
        if (lastFocus) lastFocus.focus();
      }

      document.querySelectorAll('[data-demo]').forEach((el) => {
        el.addEventListener('click', openDemo);
      });

      document.getElementById('closeModal').addEventListener('click', closeDemo);
      modal.addEventListener('click', (e) => {
        if (e.target === modal) closeDemo();
      });

      document.addEventListener('keydown', (e) => {
        if (e.key === 'Escape' && modal.classList.contains('open')) closeDemo();
      });

      const menu = document.getElementById('menuButton');
      const links = document.getElementById('navLinks');

      menu.addEventListener('click', () => {
        const open = links.classList.toggle('open');
        menu.setAttribute('aria-expanded', String(open));
        menu.setAttribute('aria-label', open ? 'Close navigation' : 'Open navigation');
      });

      links.querySelectorAll('a:not([data-demo])').forEach((a) => {
        a.addEventListener('click', () => {
          links.classList.remove('open');
          menu.setAttribute('aria-expanded', 'false');
        });
      });

      document.querySelectorAll('.toggle').forEach((toggle) => {
        toggle.addEventListener('click', () => {
          toggle.setAttribute('aria-pressed', String(toggle.getAttribute('aria-pressed') !== 'true'));
          showToast('Illustrative concept control only — no real data settings changed.');
        });
      });
    })();
